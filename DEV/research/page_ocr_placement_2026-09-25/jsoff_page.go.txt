// Throwaway: build a real overlaid page through OverlayFile with a stand-in tesseract, so the
// browser check below runs against exactly what the desktop app writes.
package main

import (
	"fmt"
	"image"
	"image/color"
	"image/png"
	"os"
	"path/filepath"

	"doc-html-translate/internal/ocr"
)

func main() {
	dir := os.Args[1]
	img := image.NewRGBA(image.Rect(0, 0, 400, 200))
	for y := 0; y < 200; y++ {
		for x := 0; x < 400; x++ {
			img.Set(x, y, color.RGBA{250, 250, 245, 255})
		}
	}
	f, _ := os.Create(filepath.Join(dir, "page.png"))
	_ = png.Encode(f, img)
	f.Close()
	html := `<!doctype html><html><head><meta charset="utf-8"></head><body><img src="page.png" width="400" height="200"></body></html>`
	page := filepath.Join(dir, "index.html")
	_ = os.WriteFile(page, []byte(html), 0o644)
	res, err := ocr.OverlayFile(os.Args[2], page, "eng", "", nil)
	fmt.Printf("%+v %v\n", res, err)
}
